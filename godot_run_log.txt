  72 | 	float white_squared = params.tonemapper_params.x;
  73 | 	
  74 | 	return color * (1.0f + color / white_squared) / (1.0f + color / params.output_max_value);

  77 | vec3 tonemap_filmic(vec3 color) {
  78 | 	
  79 | 	
  80 | 	
  81 | 	
  82 | 	const float exposure_bias = 2.0f;
  83 | 	const float A = 0.22f * exposure_bias * exposure_bias; 
  84 | 	const float B = 0.30f * exposure_bias;
  85 | 	const float C = 0.10f;
  86 | 	const float D = 0.20f;
  87 | 	const float E = 0.01f;
  88 | 	const float F = 0.30f;
  89 | 
  90 | 	vec3 color_tonemapped = ((color * (A * color + C * B) + D * E) / (color * (A * color + B) + D * F)) - E / F;
  91 | 
  92 | 	return color_tonemapped / params.tonemapper_params.x;
  93 | }

  97 | vec3 tonemap_aces(vec3 color) {
  98 | 	
  99 | 	const float exposure_bias = 1.8f;
 100 | 	const float A = 0.0245786f;
 101 | 	const float B = 0.000090537f;
 102 | 	const float C = 0.983729f;
 103 | 	const float D = 0.432951f;
 104 | 	const float E = 0.238081f;
 105 | 
 106 | 	
 107 | 	const mat3 rgb_to_rrt = mat3(
 108 | 			vec3(0.59719f * exposure_bias, 0.35458f * exposure_bias, 0.04823f * exposure_bias),
 109 | 			vec3(0.07600f * exposure_bias, 0.90834f * exposure_bias, 0.01566f * exposure_bias),
 110 | 			vec3(0.02840f * exposure_bias, 0.13383f * exposure_bias, 0.83777f * exposure_bias));
 111 | 
 112 | 	const mat3 odt_to_rgb = mat3(
 113 | 			vec3(1.60475f, -0.53108f, -0.07367f),
 114 | 			vec3(-0.10208f, 1.10813f, -0.00605f),
 115 | 			vec3(-0.00327f, -0.07276f, 1.07602f));
 116 | 
 117 | 	color *= rgb_to_rrt;
 118 | 	vec3 color_tonemapped = (color * (color + A) - B) / (color * (C * color + D) + E);
 119 | 	color_tonemapped *= odt_to_rgb;
 120 | 
 121 | 	return color_tonemapped / params.tonemapper_params.x;
 122 | }

 127 | vec3 allenwp_curve(vec3 x) {
 128 | 	
 129 | 	
 130 | 	const float awp_crossover_point = 0.18;
 131 | 	
 132 | 	
 133 | 	const float awp_shoulder_max = params.output_max_value - awp_crossover_point;
 134 | 
 135 | 	float awp_contrast = params.tonemapper_params.x;
 136 | 	float awp_toe_a = params.tonemapper_params.y;
 137 | 	float awp_slope = params.tonemapper_params.z;
 138 | 	float awp_w = params.tonemapper_params.w;
 139 | 
 140 | 	
 141 | 	vec3 s = x - awp_crossover_point;
 142 | 	vec3 slope_s = awp_slope * s;
 143 | 	s = slope_s * (1.0 + s / awp_w) / (1.0 + (slope_s / awp_shoulder_max));
 144 | 	s += awp_crossover_point;
 145 | 
 146 | 	
 147 | 	vec3 t = pow(x, vec3(awp_contrast));
 148 | 	t = t / (t + awp_toe_a);
 149 | 
 150 | 	return mix(s, t, lessThan(x, vec3(awp_crossover_point)));
 151 | }

 249 | float w0(float a) {
 250 | 	return (1.0f / 6.0f) * (a * (a * (-a + 3.0f) - 3.0f) + 1.0f);
 251 | }
 252 | 
 253 | float w1(float a) {
 254 | 	return (1.0f / 6.0f) * (a * a * (3.0f * a - 6.0f) + 4.0f);
 255 | }
 256 | 
 257 | float w2(float a) {
 258 | 	return (1.0f / 6.0f) * (a * (a * (-3.0f * a + 3.0f) + 3.0f) + 1.0f);
 259 | }

 261 | float w3(float a) {
 262 | 	return (1.0f / 6.0f) * (a * a * a);

 266 | float g0(float a) {
 267 | 	return w0(a) + w1(a);

 270 | float g1(float a) {
 271 | 	return w2(a) + w3(a);

 276 | 	return -1.0f + w1(a) / (w0(a) + w1(a));

 522 | vec3 do_fxaa(vec3 color, float exposure, vec2 uv_interp) {
 523 | 	const float EDGE_THRESHOLD_MIN = 0.0312;
 524 | 	const float EDGE_THRESHOLD_MAX = 0.125;
 525 | 	const int ITERATIONS = 12;
 526 | 	const float SUBPIXEL_QUALITY = 0.75;
 527 | 
 528 | #ifdef USE_MULTIVIEW
 529 | 	float lumaUp = rgb2luma(textureLodOffset(source_color, vec3(uv_interp, ViewIndex), 0.0, ivec2(0, 1)).xyz * exposure * params.luminance_multiplier);
 530 | 	float lumaDown = rgb2luma(textureLodOffset(source_color, vec3(uv_interp, ViewIndex), 0.0, ivec2(0, -1)).xyz * exposure * params.luminance_multiplier);
 531 | 	float lumaLeft = rgb2luma(textureLodOffset(source_color, vec3(uv_interp, ViewIndex), 0.0, ivec2(-1, 0)).xyz * exposure * params.luminance_multiplier);
 532 | 	float lumaRight = rgb2luma(textureLodOffset(source_color, vec3(uv_interp, ViewIndex), 0.0, ivec2(1, 0)).xyz * exposure * params.luminance_multiplier);
 533 | 
 534 | 	float lumaCenter = rgb2luma(color);
 535 | 
 536 | 	float lumaMin = min(lumaCenter, min(min(lumaUp, lumaDown), min(lumaLeft, lumaRight)));
 537 | 	float lumaMax = max(lumaCenter, max(max(lumaUp, lumaDown), max(lumaLeft, lumaRight)));
 538 | 
 539 | 	float lumaRange = lumaMax - lumaMin;
 540 | 
 541 | 	if (lumaRange < max(EDGE_THRESHOLD_MIN, lumaMax * EDGE_THRESHOLD_MAX)) {
 542 | 		return color;
 543 | 	}
 544 | 
 545 | 	float lumaDownLeft = rgb2luma(textureLodOffset(source_color, vec3(uv_interp, ViewIndex), 0.0, ivec2(-1, -1)).xyz * exposure * params.luminance_multiplier);
 546 | 	float lumaUpRight = rgb2luma(textureLodOffset(source_color, vec3(uv_interp, ViewIndex), 0.0, ivec2(1, 1)).xyz * exposure * params.luminance_multiplier);
 547 | 	float lumaUpLeft = rgb2luma(textureLodOffset(source_color, vec3(uv_interp, ViewIndex), 0.0, ivec2(-1, 1)).xyz * exposure * params.luminance_multiplier);
 548 | 	float lumaDownRight = rgb2luma(textureLodOffset(source_color, vec3(uv_interp, ViewIndex), 0.0, ivec2(1, -1)).xyz * exposure * params.luminance_multiplier);
 549 | 
 550 | 	float lumaDownUp = lumaDown + lumaUp;
 551 | 	float lumaLeftRight = lumaLeft + lumaRight;
 552 | 
 553 | 	float lumaLeftCorners = lumaDownLeft + lumaUpLeft;
 554 | 	float lumaDownCorners = lumaDownLeft + lumaDownRight;
 555 | 	float lumaRightCorners = lumaDownRight + lumaUpRight;
 556 | 	float lumaUpCorners = lumaUpRight + lumaUpLeft;
 557 | 
 558 | 	float edgeHorizontal = abs(-2.0 * lumaLeft + lumaLeftCorners) + abs(-2.0 * lumaCenter + lumaDownUp) * 2.0 + abs(-2.0 * lumaRight + lumaRightCorners);
 559 | 	float edgeVertical = abs(-2.0 * lumaUp + lumaUpCorners) + abs(-2.0 * lumaCenter + lumaLeftRight) * 2.0 + abs(-2.0 * lumaDown + lumaDownCorners);
 560 | 
 561 | 	bool isHorizontal = (edgeHorizontal >= edgeVertical);
 562 | 
 563 | 	float stepLength = isHorizontal ? params.pixel_size.y : params.pixel_size.x;
 564 | 
 565 | 	float luma1 = isHorizontal ? lumaDown : lumaLeft;
 566 | 	float luma2 = isHorizontal ? lumaUp : lumaRight;
 567 | 	float gradient1 = luma1 - lumaCenter;
 568 | 	float gradient2 = luma2 - lumaCenter;
 569 | 
 570 | 	bool is1Steepest = abs(gradient1) >= abs(gradient2);
 571 | 
 572 | 	float gradientScaled = 0.25 * max(abs(gradient1), abs(gradient2));
 573 | 
 574 | 	float lumaLocalAverage = 0.0;
 575 | 	if (is1Steepest) {
 576 | 		stepLength = -stepLength;
 577 | 		lumaLocalAverage = 0.5 * (luma1 + lumaCenter);
 578 | 	} else {
 579 | 		lumaLocalAverage = 0.5 * (luma2 + lumaCenter);
 580 | 	}
 581 | 
 582 | 	vec2 currentUv = uv_interp;
 583 | 	if (isHorizontal) {
 584 | 		currentUv.y += stepLength * 0.5;
 585 | 	} else {
 586 | 		currentUv.x += stepLength * 0.5;
 587 | 	}
 588 | 
 589 | 	vec2 offset = isHorizontal ? vec2(params.pixel_size.x, 0.0) : vec2(0.0, params.pixel_size.y);
 590 | 	vec3 uv1 = vec3(currentUv - offset * QUALITY(0), ViewIndex);
 591 | 	vec3 uv2 = vec3(currentUv + offset * QUALITY(0), ViewIndex);
 592 | 
 593 | 	float lumaEnd1 = rgb2luma(textureLod(source_color, uv1, 0.0).xyz * exposure * params.luminance_multiplier);
 594 | 	float lumaEnd2 = rgb2luma(textureLod(source_color, uv2, 0.0).xyz * exposure * params.luminance_multiplier);
 595 | 	lumaEnd1 -= lumaLocalAverage;
 596 | 	lumaEnd2 -= lumaLocalAverage;
 597 | 
 598 | 	bool reached1 = abs(lumaEnd1) >= gradientScaled;
 599 | 	bool reached2 = abs(lumaEnd2) >= gradientScaled;
 600 | 	bool reachedBoth = reached1 && reached2;
 601 | 
 602 | 	if (!reached1) {
 603 | 		uv1 -= vec3(offset * QUALITY(1), 0.0);
 604 | 	}
 605 | 	if (!reached2) {
 606 | 		uv2 += vec3(offset * QUALITY(1), 0.0);
 607 | 	}
 608 |

 251 | 	return (1.0f / 6.0f) * (a * (a * (-a + 3.0f) - 3.0f) + 1.0f);
 252 | }

 255 | 	return (1.0f / 6.0f) * (a * a * (3.0f * a - 6.0f) + 4.0f);

 259 | 	return (1.0f / 6.0f) * (a * (a * (-3.0f * a + 3.0f) + 3.0f) + 1.0f);

 523 | vec3 do_fxaa(vec3 color, float exposure, vec2 uv_interp) {
 524 | 	const float EDGE_THRESHOLD_MIN = 0.0312;
 525 | 	const float EDGE_THRESHOLD_MAX = 0.125;
 526 | 	const int ITERATIONS = 12;
 527 | 	const float SUBPIXEL_QUALITY = 0.75;
 528 | 
 529 | #ifdef USE_MULTIVIEW
 530 | 	float lumaUp = rgb2luma(textureLodOffset(source_color, vec3(uv_interp, ViewIndex), 0.0, ivec2(0, 1)).xyz * exposure * params.luminance_multiplier);
 531 | 	float lumaDown = rgb2luma(textureLodOffset(source_color, vec3(uv_interp, ViewIndex), 0.0, ivec2(0, -1)).xyz * exposure * params.luminance_multiplier);
 532 | 	float lumaLeft = rgb2luma(textureLodOffset(source_color, vec3(uv_interp, ViewIndex), 0.0, ivec2(-1, 0)).xyz * exposure * params.luminance_multiplier);
 533 | 	float lumaRight = rgb2luma(textureLodOffset(source_color, vec3(uv_interp, ViewIndex), 0.0, ivec2(1, 0)).xyz * exposure * params.luminance_multiplier);
 534 | 
 535 | 	float lumaCenter = rgb2luma(color);
 536 | 
 537 | 	float lumaMin = min(lumaCenter, min(min(lumaUp, lumaDown), min(lumaLeft, lumaRight)));
 538 | 	float lumaMax = max(lumaCenter, max(max(lumaUp, lumaDown), max(lumaLeft, lumaRight)));
 539 | 
 540 | 	float lumaRange = lumaMax - lumaMin;
 541 | 
 542 | 	if (lumaRange < max(EDGE_THRESHOLD_MIN, lumaMax * EDGE_THRESHOLD_MAX)) {
 543 | 		return color;
 544 | 	}
 545 | 
 546 | 	float lumaDownLeft = rgb2luma(textureLodOffset(source_color, vec3(uv_interp, ViewIndex), 0.0, ivec2(-1, -1)).xyz * exposure * params.luminance_multiplier);
 547 | 	float lumaUpRight = rgb2luma(textureLodOffset(source_color, vec3(uv_interp, ViewIndex), 0.0, ivec2(1, 1)).xyz * exposure * params.luminance_multiplier);
 548 | 	float lumaUpLeft = rgb2luma(textureLodOffset(source_color, vec3(uv_interp, ViewIndex), 0.0, ivec2(-1, 1)).xyz * exposure * params.luminance_multiplier);
 549 | 	float lumaDownRight = rgb2luma(textureLodOffset(source_color, vec3(uv_interp, ViewIndex), 0.0, ivec2(1, -1)).xyz * exposure * params.luminance_multiplier);
 550 | 
 551 | 	float lumaDownUp = lumaDown + lumaUp;
 552 | 	float lumaLeftRight = lumaLeft + lumaRight;
 553 | 
 554 | 	float lumaLeftCorners = lumaDownLeft + lumaUpLeft;
 555 | 	float lumaDownCorners = lumaDownLeft + lumaDownRight;
 556 | 	float lumaRightCorners = lumaDownRight + lumaUpRight;
 557 | 	float lumaUpCorners = lumaUpRight + lumaUpLeft;
 558 | 
 559 | 	float edgeHorizontal = abs(-2.0 * lumaLeft + lumaLeftCorners) + abs(-2.0 * lumaCenter + lumaDownUp) * 2.0 + abs(-2.0 * lumaRight + lumaRightCorners);
 560 | 	float edgeVertical = abs(-2.0 * lumaUp + lumaUpCorners) + abs(-2.0 * lumaCenter + lumaLeftRight) * 2.0 + abs(-2.0 * lumaDown + lumaDownCorners);
 561 | 
 562 | 	bool isHorizontal = (edgeHorizontal >= edgeVertical);
 563 | 
 564 | 	float stepLength = isHorizontal ? params.pixel_size.y : params.pixel_size.x;
 565 | 
 566 | 	float luma1 = isHorizontal ? lumaDown : lumaLeft;
 567 | 	float luma2 = isHorizontal ? lumaUp : lumaRight;
 568 | 	float gradient1 = luma1 - lumaCenter;
 569 | 	float gradient2 = luma2 - lumaCenter;
 570 | 
 571 | 	bool is1Steepest = abs(gradient1) >= abs(gradient2);
 572 | 
 573 | 	float gradientScaled = 0.25 * max(abs(gradient1), abs(gradient2));
 574 | 
 575 | 	float lumaLocalAverage = 0.0;
 576 | 	if (is1Steepest) {
 577 | 		stepLength = -stepLength;
 578 | 		lumaLocalAverage = 0.5 * (luma1 + lumaCenter);
 579 | 	} else {
 580 | 		lumaLocalAverage = 0.5 * (luma2 + lumaCenter);
 581 | 	}
 582 | 
 583 | 	vec2 currentUv = uv_interp;
 584 | 	if (isHorizontal) {
 585 | 		currentUv.y += stepLength * 0.5;
 586 | 	} else {
 587 | 		currentUv.x += stepLength * 0.5;
 588 | 	}
 589 | 
 590 | 	vec2 offset = isHorizontal ? vec2(params.pixel_size.x, 0.0) : vec2(0.0, params.pixel_size.y);
 591 | 	vec3 uv1 = vec3(currentUv - offset * QUALITY(0), ViewIndex);
 592 | 	vec3 uv2 = vec3(currentUv + offset * QUALITY(0), ViewIndex);
 593 | 
 594 | 	float lumaEnd1 = rgb2luma(textureLod(source_color, uv1, 0.0).xyz * exposure * params.luminance_multiplier);
 595 | 	float lumaEnd2 = rgb2luma(textureLod(source_color, uv2, 0.0).xyz * exposure * params.luminance_multiplier);
 596 | 	lumaEnd1 -= lumaLocalAverage;
 597 | 	lumaEnd2 -= lumaLocalAverage;
 598 | 
 599 | 	bool reached1 = abs(lumaEnd1) >= gradientScaled;
 600 | 	bool reached2 = abs(lumaEnd2) >= gradientScaled;
 601 | 	bool reachedBoth = reached1 && reached2;
 602 |

 604 | 		uv1 -= vec3(offset * QUALITY(1), 0.0);
 605 | 	}

 608 | 	}

 614 | 				lumaEnd1 = lumaEnd1 - lumaLocalAverage;
 615 | 			}
 616 | 			if (!reached2) {
 617 | 				lumaEnd2 = rgb2luma(textureLod(source_color, uv2, 0.0).xyz * exposure * params.luminance_multiplier);
 618 | 				lumaEnd2 = lumaEnd2 - lumaLocalAverage;
 619 | 			}
 620 | 			reached1 = abs(lumaEnd1) >= gradientScaled;
 621 | 			reached2 = abs(lumaEnd2) >= gradientScaled;
 622 | 			reachedBoth = reached1 && reached2;
 623 | 			if (!reached1) {
 624 | 				uv1 -= vec3(offset * QUALITY(i), 0.0);
 625 | 			}
 626 | 			if (!reached2) {
 627 | 				uv2 += vec3(offset * QUALITY(i), 0.0);
 628 | 			}
 629 | 			if (reachedBoth) {
 630 | 				break;
 631 | 			}
 632 | 		}
 633 | 	}
 634 | 
 635 | 	float distance1 = isHorizontal ? (uv_interp.x - uv1.x) : (uv_interp.y - uv1.y);
 636 | 	float distance2 = isHorizontal ? (uv2.x - uv_interp.x) : (uv2.y - uv_interp.y);
 637 | 
 638 | 	bool isDirection1 = distance1 < distance2;
 639 | 	float distanceFinal = min(distance1, distance2);
 640 | 
 641 | 	float edgeThickness = (distance1 + distance2);

 643 | 	bool isLumaCenterSmaller = lumaCenter < lumaLocalAverage;
 644 | 
 645 | 	bool correctVariation1 = (lumaEnd1 < 0.0) != isLumaCenterSmaller;
 646 | 	bool correctVariation2 = (lumaEnd2 < 0.0) != isLumaCenterSmaller;
 647 | 
 648 | 	bool correctVariation = isDirection1 ? correctVariation1 : correctVariation2;
 649 | 
 650 | 	float pixelOffset = -distanceFinal / edgeThickness + 0.5;
 651 | 
 652 | 	float finalOffset = correctVariation ? pixelOffset : 0.0;
 653 | 
 654 | 	float lumaAverage = (1.0 / 12.0) * (2.0 * (lumaDownUp + lumaLeftRight) + lumaLeftCorners + lumaRightCorners);
 655 | 
 656 | 	float subPixelOffset1 = clamp(abs(lumaAverage - lumaCenter) / lumaRange, 0.0, 1.0);
 657 | 	float subPixelOffset2 = (-2.0 * subPixelOffset1 + 3.0) * subPixelOffset1 * subPixelOffset1;
 658 | 
 659 | 	float subPixelOffsetFinal = subPixelOffset2 * subPixelOffset2 * SUBPIXEL_QUALITY;

 661 | 	finalOffset = max(finalOffset, subPixelOffsetFinal);
 662 | 
 663 | 	vec3 finalUv = vec3(uv_interp, ViewIndex);
 664 | 	if (isHorizontal) {
 665 | 		finalUv.y += finalOffset * stepLength;
 666 | 	} else {
 667 | 		finalUv.x += finalOffset * stepLength;
 668 | 	}
 669 | 
 670 | 	vec3 finalColor = textureLod(source_color, finalUv, 0.0).xyz * exposure * params.luminance_multiplier;
 671 | 	return finalColor;
 672 |

 674 | 	float lumaUp = rgb2luma(textureLodOffset(source_color, uv_interp, 0.0, ivec2(0, 1)).xyz * exposure * params.luminance_multiplier);
 675 | 	float lumaDown = rgb2luma(textureLodOffset(source_color, uv_interp, 0.0, ivec2(0, -1)).xyz * exposure * params.luminance_multiplier);

 677 | 	float lumaRight = rgb2luma(textureLodOffset(source_color, uv_interp, 0.0, ivec2(1, 0)).xyz * exposure * params.luminance_multiplier);

 680 | 
 681 | 	float lumaMin = min(lumaCenter, min(min(lumaUp, lumaDown), min(lumaLeft, lumaRight)));
 682 | 	float lumaMax = max(lumaCenter, max(max(lumaUp, lumaDown), max(lumaLeft, lumaRight)));
